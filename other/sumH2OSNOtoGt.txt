# This file details steps, using nco, to isolate and sum up the liquid water equiv. in
# the GIS snowpack in mm, and convert to Gt, from elm.h0 files

# copy over the glc mask file for lnd
  cp /global/cfs/cdirs/e3sm/inputdata/lnd/clm2/griddata/glcmaskdata_0.5x0.5_GIS_AIS.nc ./glcmask.nc 

# set mask values for S. hemis to 0 and rename file
  ncap2 -s 'where(LATIXY < 0.0) GLCMASK=0.0;' glcmask.nc gismask.nc

# replace dim names to be consistent w/ elm hist file dim names
  ncrename -d lsmlat,lat gismask.nc
  ncrename -d lsmlon,lon gismask.nc

# use ncrcat to extract just snow liq water thickness (in mm) from 2005 elm h0 files
  ncrcat -v H2OSNO,area 20260130.BGWCYCL20TR.ne30pg2_r05_IcoswISC30E3r5_gis1to10r02.pm-cpu.100yrTest.elm.h0.2005*nc h2osno.2005.init.nc

# append the mask field that isolates just GIS, generated above
  ncks -A -v GLCMASK gismask.nc h2osno.2005.init.nc 

# multiply the field by the mask and convert from mm of liquid to km of liquid and multiple by cell area = km^2 to get km^3 = Gt water
  ncap2 -A -s 'maskedH2OSNOkm = H2OSNO/1000/1000*area*GLCMASK' h2osno.2005.init.nc 

# sum over all cells to get integrated time series
  ncap2 -A -s 'sumMaskedH2OSNOkm = maskedH2OSNOkm.total($lat,$lon);' h2osno.2005.init.nc 

# convert timeseries in total km^3 = Gt of water to approx. global-mean SL change in mm relative to init time (divide by 360)
# (in expression below, 55984. is the initial value at time 0 in Gt)
  ncap2 -A -s 'SeaLevelChange = (sumMaskedH2OSNOkm-55984.53)/360;' h2osno.2005.init.nc
